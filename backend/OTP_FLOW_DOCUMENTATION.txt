========================================================
1. OVERVIEW
========================================================

This system implements a two-step registration process:
  Step 1: User enters username, email, password → OTP is sent to email
  Step 2: User enters OTP → Email is verified → Registration complete

The user CANNOT complete registration without verifying the OTP.

========================================================
2. HOW OTP WORKS (STEP-BY-STEP)
========================================================

STEP 1: USER FILLS REGISTRATION FORM
  → User enters: username, email, password
  → Clicks "Send OTP" button
  → Frontend calls: POST /api/auth/register

STEP 2: BACKEND PROCESSES REGISTRATION
  → Validates input (username, email, password)
  → Checks if user already exists
  → Hashes password with bcrypt
  → Generates 6-digit random OTP (e.g., 482917)
  → Sets OTP expiry to 10 minutes from now
  → Saves user to database WITH otp and otpExpires fields
  → Calls sendEmail utility to send OTP via nodemailer
  → Returns: { message: "OTP sent to your email", userId: "..." }

STEP 3: EMAIL IS SENT
  → Gmail SMTP sends email to user's email address
  → Email contains: "Your OTP code is: 482917. It is valid for 10 minutes."

STEP 4: USER ENTERS OTP
  → Frontend shows OTP input field
  → User enters the 6-digit code from email
  → Clicks "Verify OTP" button
  → Frontend calls: POST /api/auth/verify-otp

STEP 5: BACKEND VERIFIES OTP
  → Finds user by email in database
  → Checks if OTP matches
  → Checks if OTP has expired
  → If valid:
      - Clears otp and otpExpires fields from user
      - Creates session for user
      - Returns: { message: "Email verified successfully", user: {...} }
  → If invalid:
      - Returns error: "Invalid OTP"
  → If expired:
      - Returns error: "OTP has expired"

STEP 6: SUCCESS
  → Frontend shows success message
  → Redirects to /login page after 1.5 seconds

OPTIONAL: RESEND OTP
  → User clicks "Resend OTP" button
  → Frontend calls: POST /api/auth/resend-otp
  → Backend generates new OTP, new expiry
  → Sends new email
  → Old OTP is invalidated

exports.register = async (req, res) => {
  try {
    // 1. Validate input
    const { username, email, password } = await validateSignupRequest.validate(req.body);

    // 2. Check if user already exists
    const existingUser = await User.find({ $or: [{ username }, { email }] });
    if (existingUser.length > 0) {
      return res.status(409).json({ error: 'User with this username or email already exists' });
    }

    // 3. Hash password
    const hashedPassword = await bcrypt.hash(password, 10);

    // 4. Generate 6-digit OTP
    const otp = Math.floor(100000 + Math.random() * 900000).toString();
    const otpExpires = new Date(Date.now() + 10 * 60 * 1000); // 10 minutes

    // 5. Save user with OTP
    const user = await User.create({ username, email, password: hashedPassword, otp, otpExpires });

    // 6. Send OTP email
    await sendOTP(email, otp);

    // 7. Return success
    res.json({ message: 'OTP sent to your email', userId: user._id });
  } catch (error) {
    if (error.name === 'ValidationError') {
      return res.status(400).json({ error: error.errors });
    }
    res.status(400).json({ error: error.message });
  }
};


--- VERIFY OTP FUNCTION (Lines 37-66) ---
Purpose: Validates OTP, creates session, completes registration
Called by: POST /api/auth/verify-otp

exports.verifyOTP = async (req, res) => {
  try {
    const { email, otp } = req.body;

    // 1. Validate input
    if (!email || !otp) {
      return res.status(400).json({ error: 'Email and OTP are required' });
    }

    // 2. Find user
    const user = await User.findOne({ email });
    if (!user) {
      return res.status(404).json({ error: 'User not found' });
    }

    // 3. Check if OTP matches
    if (user.otp !== otp) {
      return res.status(400).json({ error: 'Invalid OTP' });
    }

    // 4. Check if OTP expired
    if (new Date() > user.otpExpires) {
      return res.status(400).json({ error: 'OTP has expired' });
    }

    // 5. Clear OTP fields
    user.otp = undefined;
    user.otpExpires = undefined;
    await user.save();

    // 6. Create session
    req.session.user = { _id: user._id, username: user.username, email: user.email, role: user.role || 'guest' };
    res.json({ message: 'Email verified successfully', user: req.session.user });
  } catch (error) {
    res.status(400).json({ error: error.message });
  }
};


--- RESEND OTP FUNCTION (Lines 68-92) ---
Purpose: Generates and sends new OTP
Called by: POST /api/auth/resend-otp

exports.resendOTP = async (req, res) => {
  try {
    const { email } = req.body;

    if (!email) {
      return res.status(400).json({ error: 'Email is required' });
    }

    const user = await User.findOne({ email });
    if (!user) {
      return res.status(404).json({ error: 'User not found' });
    }

    // 1. Generate new OTP
    const otp = Math.floor(100000 + Math.random() * 900000).toString();
    const otpExpires = new Date(Date.now() + 10 * 60 * 1000);

    // 2. Save new OTP
    user.otp = otp;
    user.otpExpires = otpExpires;
    await user.save();

    // 3. Send email
    await sendOTP(email, otp);

    res.json({ message: 'New OTP sent to your email' });
  } catch (error) {
    res.status(400).json({ error: error.message });
  }
};

const BASE_URL = import.meta.env.VITE_API_URL || 'http://localhost:5000/api';

const getHeaders = () => ({
  'Content-Type': 'application/json',
});

const handleResponse = async (resp) => {
  let data;
  try {
    data = await resp.json();
  } catch {
    data = null;
  }
  if (!resp.ok) {
    throw new Error(data?.error || data?.message || 'API error');
  }
  return data;
};

// REGISTER - Step 1: Send OTP
export const register = async (payload) => {
  const resp = await fetch(`${BASE_URL}/auth/register`, {
    method: 'POST',
    headers: getHeaders(),
    body: JSON.stringify(payload),
    credentials: 'include',
  });
  return handleResponse(resp);
};

// VERIFY OTP - Step 2: Verify OTP
export const verifyOTP = async (payload) => {
  const resp = await fetch(`${BASE_URL}/auth/verify-otp`, {
    method: 'POST',
    headers: getHeaders(),
    body: JSON.stringify(payload),
    credentials: 'include',
  });
  return handleResponse(resp);
};

// RESEND OTP - Optional: Get new OTP
export const resendOTP = async (payload) => {
  const resp = await fetch(`${BASE_URL}/auth/resend-otp`, {
    method: 'POST',
    headers: getHeaders(),
    body: JSON.stringify(payload),
    credentials: 'include',
  });
  return handleResponse(resp);
};

========================================================
7. DATA FLOW EXAMPLE
========================================================

EXAMPLE: User "john" registers with email "john@example.com"

1. USER fills form:
   username: "john"
   email: "john@example.com"
   password: "secret123"

2. FRONTEND calls: POST /api/auth/register
   Request body: { username: "john", email: "john@example.com", password: "secret123" }

3. BACKEND processes:
   → Hashes password: "$2a$10$..."
   → Generates OTP: "482917"
   → Sets expiry: 2025-04-05T10:10:00.000Z (10 minutes)
   → Saves to MongoDB:
     {
       _id: "abc123",
       username: "john",
       email: "john@example.com",
       password: "$2a$10$...",
       otp: "482917",
       otpExpires: "2025-04-05T10:10:00.000Z",
       role: "guest"
     }
   → Sends email via Gmail SMTP

4. USER receives email:
   Subject: "Your OTP Code"
   Body: "Your OTP code is: 482917. It is valid for 10 minutes."

5. USER enters OTP: "482917"

6. FRONTEND calls: POST /api/auth/verify-otp
   Request body: { email: "john@example.com", otp: "482917" }

7. BACKEND verifies:
   → Finds user by email
   → Checks: otp === "482917" ✓
   → Checks: now < otpExpires ✓
   → Clears otp and otpExpires fields
   → Creates session
   → Returns: { message: "Email verified successfully", user: {...} }

8. FRONTEND shows success → Redirects to /login
